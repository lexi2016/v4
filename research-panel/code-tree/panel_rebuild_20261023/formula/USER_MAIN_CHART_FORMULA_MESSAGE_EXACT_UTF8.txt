BBI:(MA(CLOSE,3)+MA(CLOSE,6)+MA(CLOSE,12)+MA(CLOSE,24))/4,COLORBLUE;
白线:EMA(EMA(CLOSE,10),10),COLORFFFFFF,LINETHICK2;
黄线:(MA(CLOSE,14)+MA(CLOSE,28)+MA(CLOSE,57)+MA(CLOSE,114))/4,COLORFBCD2D,LINETHICK2;
MA60:MA(CLOSE,60),COLOR808080,LINETHICK1;
向上缺口:=LOW>REF(HIGH,1);
昨日守线:=IF(向上缺口,REF(HIGH,1),MIN(LOW,REF(CLOSE,1)));
守线:IF(CURRBARSCOUNT<=2,REF(昨日守线,1),DRAWNULL),COLORFFFFFF;
今天:=CURRBARSCOUNT=1;
DRAWTEXT(今天 AND CLOSE>=REF(昨日守线,1),LOW*0.992,'守线'),COLORFFFFFF;
DRAWTEXT(今天 AND CLOSE<REF(昨日守线,1),LOW*0.992,'破线'),COLORGREEN;
DRAWICON(今天 AND CLOSE<REF(昨日守线,1),HIGH*1.020,2);
RSV:=(CLOSE-LLV(LOW,9))/(HHV(HIGH,9)-LLV(LOW,9))*100;
K:=SMA(RSV,3,1);
D:=SMA(K,3,1);
J:=3*K-2*D;
今日涨幅:=(CLOSE/REF(CLOSE,1)-1)*100;
昨日涨幅:=(REF(CLOSE,1)/REF(CLOSE,2)-1)*100;
昨日放量:=REF(VOL,1)>MA(REF(VOL,2),4)*1.15;
B1:=BBI>白线 AND 白线>黄线 AND J<=13;
DRAWTEXT(B1,LOW*0.975,'B1'),COLORRED;
B2:=COUNT(J<13,3)>0 AND VOL>MA(REF(VOL,1),4)*1.25 AND 今日涨幅>4 AND J<70 AND 白线>黄线;
DRAWTEXT(B2,LOW*0.955,'B2'),COLORRED;
VOLR:=VOL/REF(VOL,1);
缩半量:=VOLR<0.6;
平量:=VOLR>0.9;
真突破:=今日涨幅>0 OR (今日涨幅>-3 AND (缩半量 OR 平量));
B3:=REF(J,2)<20 AND 昨日放量 AND 昨日涨幅>3 AND 白线>黄线 AND 真突破;
DRAWTEXT(B3,LOW*0.935,'B3'),COLORRED;
近20日高点:=HHV(HIGH,20);
破前高:=CLOSE>REF(近20日高点,1) AND VOL>REF(VOL,1)*0.8 AND 今日涨幅>4;
突破:=破前高 AND COUNT(破前高,8)=1;
DRAWTEXT(突破,LOW*0.915,'突破'),COLORRED;
昨日突破前高:=REF(CLOSE,1)>=REF(近20日高点,2) AND 昨日放量 AND 昨日涨幅>4;
突破接十字星:=COUNT(昨日突破前高,8)=1 AND 昨日突破前高 AND 今日涨幅<4 AND 真突破;
DRAWTEXT(突破接十字星,HIGH*1.055,'突破十字'),COLORRED;
四分之三量:=VOLR<=0.9 AND VOLR>=0.6 AND 昨日放量 AND 昨日涨幅>4;
四分之三阴量线:=COUNT(昨日突破前高,8)=1 AND 昨日突破前高 AND 今日涨幅<0 AND 四分之三量;
DRAWTEXT(四分之三阴量线,HIGH*1.075,'3/4量'),COLORRED;
短期:=100*(CLOSE-LLV(LOW,3))/(HHV(CLOSE,3)-LLV(LOW,3));
长期:=100*(CLOSE-LLV(LOW,21))/(HHV(CLOSE,21)-LLV(LOW,21));
单针下20:=长期>=75 AND 短期<=25;
DRAWTEXT(单针下20,LOW*0.975,'单针底'),COLORRED;
前日低点:=MIN(REF(CLOSE,2),REF(LOW,1));
破前低:=CLOSE/前日低点<0.99;
DRAWTEXT(破前低,LOW*0.955,'破前低'),COLORRED;
黄线下:=COUNT(J<13,10)>0 AND REF(CLOSE,2)>黄线 AND REF(CLOSE,1)<黄线 AND CLOSE<黄线 AND 白线>黄线;
DRAWTEXT(黄线下,LOW*0.935,'黄线下'),COLORRED;
成本价:=REF(CLOSE,BARSLAST(J<13));
二十日内B1:=BARSLAST(B1)<=20;
飞1:=二十日内B1 AND HIGH/成本价-1>0.15 AND REF(HIGH,1)/成本价-1<0.15;
DRAWTEXT(飞1,HIGH*1.095,'飞1'),COLORRED;
二十五日内B1:=BARSLAST(B1)<=25;
飞2:=二十五日内B1 AND HIGH/成本价-1>0.25 AND REF(HIGH,1)/成本价-1<0.25;
DRAWTEXT(飞2,HIGH*1.115,'飞2'),COLORRED;
DIF:=EMA(CLOSE,12)-EMA(CLOSE,26);
DEA:=EMA(DIF,9);
金叉:=CROSS(DIF,DEA);
零下金次:=COUNT(金叉,BARSLAST(DEA>=0));
零下二金:=金叉 AND DEA<0 AND COUNT(零下金次=2,21)=1;
零上金次:=COUNT(金叉,BARSLAST(DEA<=0));
零上二金:=金叉 AND DEA>0 AND 零上金次>=2;
DRAWTEXT(零下二金,LOW*0.900,'M2-'),COLORYELLOW;
DRAWTEXT(零上二金,LOW*0.880,'M2+'),COLORRED;
死叉:=白线<黄线 AND REF(白线,1)>REF(黄线,1);
卖点:=今天 AND CLOSE<REF(昨日守线,1) OR 死叉 OR 破前低;
一级买点:=B3 OR 零上二金;
二级买点:=(B1 OR B2 OR 突破 OR 突破接十字星 OR 零下二金) AND 一级买点=0;
DRAWICON(一级买点,LOW*0.905,1);
DRAWTEXT(一级买点,LOW*0.890,'强买'),COLORFF00FF;
DRAWICON(二级买点,LOW*0.930,1);
DRAWTEXT(二级买点,LOW*0.915,'买点'),COLORFFFFFF;
DRAWICON(卖点,HIGH*1.020,2);
DRAWTEXT(卖点,HIGH*1.040,'卖点'),COLORGREEN;
