// 关注股公告 / 新闻聚合：从本地关注列表取代码，复用个股接口批量拉取、按时间倒序合并。
// 只做公开信息聚合，标的均为用户自己关注列表里的，不预置、不推荐。
interface FeedRow { code: string; name: string; when: string; title: string; meta?: string; url?: string }
const MAX_ROWS = 60;

function WatchlistFeed({ kind }: { kind: "filings" | "news" }) {
  const [codes, setCodes] = useState<string[]>(loadWatch);
  const [rows, setRows] = useState<FeedRow[]>([]);
  const [loading, setLoading] = useState(false);
  const [err, setErr] = useState<string | null>(null);
  const [depNote, setDepNote] = useState<string | null>(null);

  const load = useCallback(async (cs: string[]) => {
    if (!cs.length) { setRows([]); return; }
    setLoading(true); setErr(null); setDepNote(null);
    try {
      const symbols = cs.map((c) => normalizeMarketSymbol(c)).filter((s): s is MarketSymbol => !!s);
      const aSymbols = symbols.filter((s) => s.market === "A");
      const hkSymbols = symbols.filter((s) => s.market === "HK");
      // 股名：A 股批量读取；港股只走现有 globalStock adapter，绝不投递 A 股接口。
      const nameOf: Record<string, string> = {};
      if (aSymbols.length) {
        try {
          const quotes = await api.quote(aSymbols.map((s) => s.code).join(","));
          for (const s of aSymbols) if (quotes[s.canonical]?.name) nameOf[s.canonical] = quotes[s.canonical].name;
        } catch { /* 忽略：无股名不影响公告/新闻 */ }
      }
      await Promise.all(hkSymbols.map(async (s) => {
        try {
          const q = await api.globalStock(s.query);
          if (q.name) nameOf[s.canonical] = q.name;
        } catch { /* 港股行情缺失时保留规范化代码 */ }
      }));

      const out: FeedRow[] = [];
      if (kind === "filings") {
        const res = await Promise.all(
          aSymbols.map((s) => api.announcements(s.code).then((a) => ({ c: s.canonical, a })).catch(() => ({ c: s.canonical, a: [] as Announcement[] }))),
        );
        for (const { c, a } of res)
          for (const x of a)
            out.push({ code: c, name: nameOf[c] || c, when: x.date, title: x.title.replace(/^[^:：]*[:：]/, ""), meta: x.type, url: x.url });
        for (const s of hkSymbols)
          out.push({ code: s.canonical, name: nameOf[s.canonical] || s.canonical, when: "", title: "港股公告数据源未接入", meta: "缺源" });
      } else {
        let dep: string | null = null;
        const res = await Promise.all(
          aSymbols.map((s) =>
            api.news(s.code).then((n) => ({ c: s.canonical, n })).catch((e) => {
              if (e instanceof ApiError && e.status === 501) dep = e.message;
              return { c: s.canonical, n: [] as NewsItem[] };
            }),
          ),
        );
        for (const { c, n } of res)
          for (const x of n)
            out.push({ code: c, name: nameOf[c] || c, when: x.发布时间 || "", title: x.新闻标题 || "", url: x.新闻链接 });
        for (const s of hkSymbols)
          out.push({ code: s.canonical, name: nameOf[s.canonical] || s.canonical, when: "", title: "港股新闻数据源未接入", meta: "缺源" });
        if (dep && out.length === 0) setDepNote(dep);
      }
      // 按真实时间倒序：多新闻源的时间字符串格式不统一（有无秒/斜杠日期），字典序会排乱
      const ts = (s: string) => {
        const raw = (s || "").trim();
        let t = Date.parse(raw);
        if (Number.isNaN(t)) t = Date.parse(raw.replace(" ", "T"));
        return Number.isNaN(t) ? 0 : t;
      };
      out.sort((p, q) => ts(q.when) - ts(p.when));
      setRows(out.slice(0, MAX_ROWS));
    } catch (e) {
      setErr(e instanceof ApiError ? e.message : "加载失败");
